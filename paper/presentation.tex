\section{Introduction \& Motivation}

\begin{frame}{The Challenge \& Our Approach}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Key Question}
      
      \textit{Do complex transformer models outperform simpler baselines on daily stock data?}
      
      \vspace{0.5cm}
      
      \textbf{Why is this hard?}
      \begin{itemize}
        \item Noisy, low signal-to-noise ratio
        \item Regime-dependent patterns
        \item Market efficiency limits
      \end{itemize}
    \end{column}
    
    \begin{column}{0.45\textwidth}
      \textbf{Our Contribution}
      \begin{itemize}
        \item Fair comparison: identical features across all models
        \item Strict temporal validation
        \item Multiple horizons (1, 5, 21 days)
        \item Regime \& event analysis
        \item Dow 30, 2016--2024
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Three Core Hypotheses}
  \begin{block}{H1: Model Class Advantage}
    \textbf{Hypothesis:} Temporal Fusion Transformer (TFT) will outperform baselines at $h=21$ days\\
    \textbf{Rationale:} Attention mechanisms should handle long-term dependencies better\\
    \textbf{Criteria:} $\geq$2\,pp improvement in DA or $\geq$5\% reduction in RMSE
  \end{block}
  
  \begin{block}{H2: Event Features Add Value}
    \textbf{Hypothesis:} Adding earnings features improves DA without worsening RMSE\\
    \textbf{Rationale:} Post-earnings-announcement drift is well-documented\\
    \textbf{Criteria:} DA $\uparrow$ by $\geq$2\,pp while RMSE stable
  \end{block}
  
  \begin{block}{H3: Regime Dependence}
    \textbf{Hypothesis:} Model rankings differ between bear and bull markets\\
    \textbf{Rationale:} Adaptive Markets Hypothesis suggests pattern instability\\
    \textbf{Criteria:} Different best-performing models across 2022 bear vs. 2023--24 rally
  \end{block}
\end{frame}

\begin{frame}{Literature \& Research Gaps}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Existing Approaches}
      \begin{itemize}\small
        \item \textbf{Classical:} ARIMA, VAR—assume linearity/stationarity
        \item \textbf{Tree ensembles:} XGBoost, Random Forest—strong on cross-sections
        \item \textbf{RNNs:} LSTM/GRU capture temporal patterns
        \item \textbf{Transformers:} TFT, PatchTST—excel with scale \& multimodal data
        \item \textbf{Recent:} TimesFM (zero-shot), BloombergGPT (text+price)
      \end{itemize}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \textbf{Key Gaps}
      \begin{itemize}\small
        \item Few apples-to-apples comparisons across model classes
        \item Earnings/event features underused in deep models
        \item Look-ahead bias risks in evaluation
        \item Limited regime analysis
      \end{itemize}
      
      \vspace{0.3cm}
      
      \textbf{Our Contribution}
      \begin{itemize}\small
        \item Fair comparison: identical features
        \item Earnings + technical inputs
        \item Strict temporal validation
        \item Regime \& event breakdown
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Methodology}

\begin{frame}{Experimental Design Overview}
  \begin{center}
    \includegraphics[width=0.9\textwidth]{figs/experiment_timeline.png}
  \end{center}
  
  \textbf{Data:} Dow 30 stocks (fixed as of 2018)\\[0.2cm]
  \textbf{Timeline:}
  \begin{itemize}
    \item \textcolor{blue}{Train:} 2016--2019 (4 years)
    \item \textcolor{orange}{Validation:} 2020 (COVID year, held out)
    \item \textcolor{red}{Test:} 2021--2024 (4 years, multiple regimes)
  \end{itemize}
  
  \vspace{0.3cm}
  
  \textbf{Horizons:} $h \in \{1, 5, 21\}$ trading days\\[0.2cm]
  \textbf{Target:} Log returns $r_{t\to t+h} = \ln(P_{t+h}) - \ln(P_t)$
\end{frame}

\begin{frame}{Data \& Features}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Features}
      \begin{itemize}
        \item Technical: momentum, MA, volatility, RSI, MACD
        \item Earnings: surprise \%, days to/from announcement
        \item Event windows ($\pm$3 days)
      \end{itemize}
      
      \vspace{0.3cm}
      
      \textbf{Leakage Controls}
      \begin{itemize}
        \item Forward-only validation
        \item 5-day embargos
        \item Scaling fit on train only
      \end{itemize}
    \end{column}
    
    \begin{column}{0.45\textwidth}
      \textbf{Data Sources}
      \begin{itemize}
        \item Prices: yfinance
        \item Earnings: API Ninjas
        \item Macro: FRED API
      \end{itemize}
      
      \vspace{0.3cm}
      
      \textbf{Training}
      \begin{itemize}
        \item Pooled across all 30 stocks
        \item Shared parameters
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Models Tested}
  \small
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \textbf{Tabular Models}
      \begin{itemize}\scriptsize
        \item \textbf{Ridge Regression}
        \begin{itemize}
          \item Linear baseline
          \item L2 regularization
          \item $\alpha$ tuned on validation
        \end{itemize}
        
        \item \textbf{Random Forest}
        \begin{itemize}
          \item 100 trees, depth 5
          \item Strong tabular learner
        \end{itemize}
        
        \item \textbf{XGBoost}
        \begin{itemize}
          \item Gradient boosting
          \item Depth 3, lr 0.1
        \end{itemize}
      \end{itemize}
      
      \vspace{0.2cm}
      \scriptsize
      \textit{Trained separately per horizon}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \textbf{Sequence Models}
      \begin{itemize}
        \item LSTM (64 hidden, 60-day lookback)
        \item GRU (simpler than LSTM)
        \item TFT (4 attn heads, $\sim$50k params)
      \end{itemize}
      
      \vspace{0.3cm}
      \scriptsize
      \textit{Predict all horizons simultaneously}
      
      \textbf{Metrics:} RMSE, $\rTwo$, Directional Accuracy (DA)
      \begin{itemize}
        \item DA = fraction of correct directional predictions
        \item 50\% = random; $>$50\% = predictive value
      \end{itemize}
    \end{column}
  \end{columns}
  
\end{frame}

\section{Results}

\begin{frame}{Main Results: Performance by Horizon}
  \begin{center}
    \scriptsize
    \begin{tabular}{llccc}
      \toprule
      \textbf{Model} & \textbf{$h$} & \textbf{RMSE} & \textbf{$\rTwo$} & \textbf{DA} \\
      \midrule

      Ridge & 1 & 0.0163 & -0.018 & 50.0\% \\
      \midrule

      \textbf{LSTM} & 5 & 0.0578 & -0.002 & \textbf{55.4\%} \\
      GRU & 5 & 0.0579 & -0.003 & 54.2\% \\
      \midrule
      \multicolumn{5}{c}{\textit{$h=21$ (one month)}} \\
      \textbf{LSTM} & 21 & \textbf{0.0508} & -0.002 & \textbf{55.0\%} \\
      GRU & 21 & 0.0509 & -0.002 & 53.9\% \\
      TFT & 21 & 0.0510 & -0.007 & 53.9\% \\
      Ridge & 21 & 0.0742 & -0.028 & 52.8\% \\
      \bottomrule
    \end{tabular}
  \end{center}
  
  \vspace{0.2cm}
  
  \begin{alertblock}{Key Takeaway}
    \small
    \textbf{LSTM} wins at $h=1$ \& $h=21$. \textbf{Ridge} dominates at $h=5$. TFT competitive but not superior.
  \end{alertblock}
\end{frame}

\begin{frame}{Key Observations}
  \begin{center}
    \includegraphics[width=0.45\textwidth]{../comparison/artifacts/figures/fig1_da_by_horizon.png}

    \item At $h=21$: LSTM reaches 55\% DA
    \item TFT competitive but not superior
    \item Negative $\rTwo$ values normal in finance (low signal/noise)

\section{Discussion \& Conclusions}

\begin{frame}{Why Simple Models Win}
  \begin{enumerate}
    \item \textbf{Data constraints}: Small panel (30 stocks, daily)—limited for deep learning
    
    \item \textbf{Low signal-to-noise}: Short-term patterns quasi-linear
    
    \item \textbf{Model capacity}: TFT ($\sim$50k params) vs. LSTM ($\sim$25k)—overfitting risk
    
    \item \textbf{Feature engineering}: Technical indicators capture most signal
  \end{enumerate}
  
  \vspace{0.5cm}
  
  \textbf{When do transformers help?}
  \begin{itemize}
    \item Large datasets (100B+ time points)
    \item Multimodal inputs (text + prices)
    \item Pretraining / transfer learning
  \end{itemize}
\end{frame}



\begin{frame}{Key Takeaways \& Future Directions}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Main Findings}
      \begin{enumerate}
        \item Simple models competitive (LSTM/Ridge)
        \item Earnings features valuable (+7.5\,pp DA)
        \item Performance regime-dependent
        \item Transformers need scale
      \end{enumerate}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \textbf{Future Work}
      \begin{itemize}
        \item Multimodal integration (text + prices)
        \item Larger scale (S\&P 500, intraday)
        \item Advanced architectures (PatchTST)
        \item Economic evaluation
      \end{itemize}
    \end{column}
  \end{columns}
  
  \vspace{0.5cm}
  
  \begin{alertblock}{Bottom Line}
    Start simple, engineer features carefully, validate rigorously. Transformers excel with scale and multimodal data.
  \end{alertblock}
\end{frame}
